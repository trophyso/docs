```java Java
CreateRewardsResponse response = client.admin().rewards().create(
  List.of(
    CreateRewardRequestItem.builder()
      .name("Welcome bonus")
      .key("welcome-bonus")
      .type(CreateRewardRequestItemType.POINTS)
      .pointsSystemId("points-system-uuid")
      .points(100)
      .triggers(List.of(
        CreateRewardTriggerRequestItem.builder()
          .type(CreateRewardTriggerRequestItemType.ACHIEVEMENT)
          .achievementId("achievement-uuid")
          .status(CreateRewardTriggerRequestItemStatus.ACTIVE)
          .build()
      ))
      .build()
  )
);
```
